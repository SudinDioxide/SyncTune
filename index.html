📋 1-Click Database SQL & Storage Setup Guide

Step 1: In your Supabase Dashboard, click SQL Editor, paste the script below, and click Run:

create table if not exists public.tracks (
  id uuid default gen_random_uuid() primary key,
  title text not null,
  artist text default 'Unknown Artist',
  album text default 'Single',
  duration numeric default 0,
  audio_url text not null,
  cover_url text,
  synced_lyrics text,
  size_bytes bigint default 0,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.tracks enable row level security;
create policy "Allow public read" on public.tracks for select using (true);
create policy "Allow public insert" on public.tracks for insert with check (true);
create policy "Allow public delete" on public.tracks for delete using (true);

Step 2: Go to Storage -> New Bucket -> Name it songs -> Turn Public bucket: ON.
